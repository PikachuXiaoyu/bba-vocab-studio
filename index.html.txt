    function loadState() {
      try {
        const raw = localStorage.getItem(STORAGE_KEY);
        if (!raw) return getDefaultState();
        const parsed = JSON.parse(raw);
        return {
          ...getDefaultState(),
          ...parsed,
          wordStats: parsed.wordStats || {},
          chapterProgress: parsed.chapterProgress || {},
          unlockedChapters: Array.isArray(parsed.unlockedChapters) && parsed.unlockedChapters.length > 0
            ? parsed.unlockedChapters
            : ['ch01'],
          completedChapters: Array.isArray(parsed.completedChapters) ? parsed.completedChapters : []
        };
      } catch (e) {
        console.error('读取进度失败，已恢复默认状态:', e);
        return getDefaultState();
      }
    }

      try {

      }
    }

    /**
     * ============================================================================
     * 三、 英语语音朗读引擎 (WEB SPEECH API · 优先原生英音 + 语速 + 防叠加)
     * ============================================================================
     */
    let selectedEnglishVoice = null;

    function initSpeechVoices() {
      if (!('speechSynthesis' in window)) return;
      const voices = window.speechSynthesis.getVoices();
      if (!voices || voices.length === 0) return;

      // 优先筛选高品质原生英语语音包（en-US / en-GB）
      const enVoices = voices.filter(v => v.lang && (v.lang.startsWith('en-US') || v.lang.startsWith('en-GB') || v.lang.startsWith('en')));
      selectedEnglishVoice =
        enVoices.find(v => /Google US English|Microsoft Aria|Microsoft Guy|Samantha|Daniel|Alex/i.test(v.name)) ||
        enVoices.find(v => v.lang === 'en-US') ||
        enVoices[0] ||
        null;
    }

    if ('speechSynthesis' in window) {
      initSpeechVoices();
      window.speechSynthesis.onvoiceschanged = initSpeechVoices;
    }

    /**
     * 播放英语发音（严格执行 cancel() 防叠加）
     * @param {string} text - 要朗读的英文单词或例句
     */
    function speakText(text) {
      if (!('speechSynthesis' in window)) {
        showToast('当前浏览器不支持原生语音合成功能');
        return;
      }
      // 关键防叠加控制：立即取消当前正在播放或排队的语音
      window.speechSynthesis.cancel();

      const utterance = new SpeechSynthesisUtterance(text);
      utterance.lang = 'en-US';
      if (selectedEnglishVoice) {
        utterance.voice = selectedEnglishVoice;
      }
      utterance.rate = appState.speechRate || 1.0;
      utterance.pitch = 1.0;

      window.speechSynthesis.speak(utterance);
    }

    }

      }
    }

    /**
     * ============================================================================
     * 四、 文本规范化、洗牌算法、拼写对比 Diff 与双维度错词计数
     * ============================================================================
     */

    /**
     * 严谨的默写文本归一化：
     * 1. 忽略首尾空格 (.trim())
     * 2. 不区分大小写 (.toLowerCase())
     * 3. 连续多个空格合并为单个空格 (.replace(/\s+/g, ' '))
     * 4. 单词/词组内部的单个合法空格严格保留（例如 "organizational goals" 不可写成 "organizationalgoals"）
     */
    function normalizeDictationText(str) {
      if (typeof str !== 'string') return '';
      return str.trim().replace(/\s+/g, ' ').toLowerCase();
    }

    // Fisher-Yates 随机洗牌算法
    function shuffleArray(arr) {
      const copy = [...arr];
      for (let i = copy.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [copy[i], copy[j]] = [copy[j], copy[i]];
      }
      return copy;
    }

    /**
     * 生成字符级拼写差异高亮 HTML（用于默写答错时的直观反馈）
     */
    function buildSpellingDiffHtml(userRaw, targetStandard) {
      const userNorm = userRaw.trim().replace(/\s+/g, ' ');
      const target = targetStandard.trim();
      const maxLen = Math.max(userNorm.length, target.length);
      let userHtml = '';

      for (let i = 0; i < maxLen; i++) {
        const uChar = userNorm[i];
        const tChar = target[i];
        if (uChar === undefined) {
          // 漏写字符
          userHtml += `<span class="inline-block px-1 mx-0.5 rounded bg-amber-100 text-amber-800 border border-amber-300 font-mono text-xs" title="此处漏写">_</span>`;
        } else if (tChar === undefined || uChar.toLowerCase() !== tChar.toLowerCase()) {
          // 错写或多写字符
          const displayChar = uChar === ' ' ? '␣' : escapeHtml(uChar);
          userHtml += `<span class="inline-block px-1 mx-0.5 rounded bg-rose-100 text-rose-700 line-through font-bold font-mono">${displayChar}</span>`;
        } else {
          // 正确字符
          const displayChar = uChar === ' ' ? '&nbsp;' : escapeHtml(uChar);
          userHtml += `<span class="text-emerald-700 font-semibold font-mono">${displayChar}</span>`;
        }
      }

      return `
        <div class="mt-3 p-3.5 rounded-xl bg-slate-50 border border-slate-200 text-sm space-y-2">
          <div class="flex flex-wrap items-center gap-2">
            <span class="text-xs font-semibold text-slate-500 uppercase tracking-wider w-20">你的输入：</span>
            <div class="tracking-wide">${userHtml || '<span class="text-slate-400 italic">（空）</span>'}</div>
          </div>
          <div class="flex flex-wrap items-center gap-2">
            <span class="text-xs font-semibold text-slate-500 uppercase tracking-wider w-20">标准拼写：</span>
            <div class="font-mono font-bold text-emerald-700 tracking-wide">${escapeHtml(targetStandard)}</div>
          </div>
          <p class="text-[11px] text-slate-500 pt-1">提示：红色删除线为拼写不符或多余字符，黄色下划线 <code class="bg-amber-100 px-1 rounded">_</code> 为漏写字符；短语内部的空格必须保留。</p>
        </div>
      `;
    }

    /**
     * 更新单词双维度错误计数器（仅在首次提交该独立题目时调用）
     * - 答错：totalErrors + 1，consecutiveErrors + 1
     * - 答对：totalErrors 保持不变（绝不清零），consecutiveErrors 重置为 0
     */
    function recordWordAttempt(wordId, isCorrect) {
      if (!appState.wordStats[wordId]) {
        appState.wordStats[wordId] = { totalErrors: 0, consecutiveErrors: 0 };
      }
      if (isCorrect) {
        appState.wordStats[wordId].consecutiveErrors = 0;
      } else {
        appState.wordStats[wordId].totalErrors += 1;
        appState.wordStats[wordId].consecutiveErrors += 1;
      }
      saveState();
      return appState.wordStats[wordId];
    }

    function getWordErrorStat(wordId) {
      return appState.wordStats[wordId] || { totalErrors: 0, consecutiveErrors: 0 };
    }

    /**
     * ============================================================================
     * 五、 考核题目工厂与断点状态构建器 (ASSESSMENT BUILDER)
     * ============================================================================
     */

    /**
     * 为指定单词列表构建三阶段完整考核题库（选项均在创建时洗牌并保存至状态，保证断点恢复一致性）
     * @param {Array} targetWords - 本轮要考核的单词对象数组
     * @param {Array} distractorPoolWords - 用于抽取同章/同领域干扰项的词池
     */
    function buildThreeStageQuestions(targetWords, distractorPoolWords) {
      // 阶段 1：四选一认词（优先从同章其他词汇释义中抽取干扰项）
      const stage1 = shuffleArray(targetWords).map((w, idx) => {
        const samePoolMeanings = distractorPoolWords
          .filter(other => other.id !== w.id && other.meaning !== w.meaning)
          .map(other => other.meaning);

        let chosenDistractors = shuffleArray(samePoolMeanings).slice(0, 3);
        // 若词数不足 4 个，使用预设的专业领域 distractors 兜底补足
        if (chosenDistractors.length < 3 && Array.isArray(w.distractors)) {
          for (const d of w.distractors) {
            if (chosenDistractors.length >= 3) break;
            if (d !== w.meaning && !chosenDistractors.includes(d)) {
              chosenDistractors.push(d);
            }
          }
        }

        const options = shuffleArray([w.meaning, ...chosenDistractors]);
        return {
          uid: `s1_${w.id}_${idx}`,
          stage: 1,
          wordId: w.id,
          options,
          correctAnswer: w.meaning,
          submitted: false,
          userAnswer: null,
          isCorrect: null
        };
      });

      // 阶段 2：听音 + 首字母 + 中文释义默写
      const stage2 = shuffleArray(targetWords).map((w, idx) => ({
        uid: `s2_${w.id}_${idx}`,
        stage: 2,
        wordId: w.id,
        correctAnswer: w.word,
        submitted: false,
        userAnswer: '',
        isCorrect: null
      }));

      // 阶段 3：简单场景应用（商科语境选词）
      const stage3 = shuffleArray(targetWords).map((w, idx) => {
        const samePoolWords = distractorPoolWords
          .filter(other => other.id !== w.id && other.word !== w.word)
          .map(other => other.word);

        let chosenWordDistractors = shuffleArray(samePoolWords).slice(0, 3);
        // 若词池不足 4 个，从全局其他章节补充商科词汇作为干扰项
        if (chosenWordDistractors.length < 3) {
          const globalWords = CHAPTERS_DATA.flatMap(c => c.words)
            .filter(other => other.id !== w.id && !chosenWordDistractors.includes(other.word))
            .map(other => other.word);
          for (const gw of shuffleArray(globalWords)) {
            if (chosenWordDistractors.length >= 3) break;
            chosenWordDistractors.push(gw);
          }
        }

        const options = shuffleArray([w.word, ...chosenWordDistractors]);
        return {
          uid: `s3_${w.id}_${idx}`,
          stage: 3,
          wordId: w.id,
          options,
          correctAnswer: w.word,
          submitted: false,
          userAnswer: null,
          isCorrect: null
        };
      });

      return { 1: stage1, 2: stage2, 3: stage3 };
    }

    /**
     * 获取或初始化某一章节的学习/考核进度（支持断点续学）
     */
    function ensureChapterProgress(chapterId) {
      if (!appState.chapterProgress[chapterId]) {
        const chapter = findChapterById(chapterId);
        appState.chapterProgress[chapterId] = {
          mode: 'flashcards', // 'flashcards' | 'assessment' | 'summary'
          flashcardIndex: 0,
          viewedCards: [0],
          currentStage: 1, // 1 | 2 | 3
          currentQuestionIndex: 0,
          questions: buildThreeStageQuestions(chapter.words, chapter.words)
        };
        saveState();
      }
      return appState.chapterProgress[chapterId];
    }

    /**

     * ============================================================================
     */

    function navigateHome() {
      if ('speechSynthesis' in window) window.speechSynthesis.cancel();
      currentRoute.view = 'home';
      render();
    }

    function openChapter(chapterId, forceMode = null) {
      if (!appState.unlockedChapters.includes(chapterId)) {
        showToast('请先完成前一章节的考核以解锁本章节');
        return;
      }
      const prog = ensureChapterProgress(chapterId);
      if (forceMode === 'flashcards') {
        prog.mode = 'flashcards';
        saveState();
      } else if (forceMode === 'assessment') {
        if (prog.mode === 'summary') {
          // 如果已完成过一轮想再考一次，重新生成题组
          const chapter = findChapterById(chapterId);
          prog.currentStage = 1;
          prog.currentQuestionIndex = 0;
          prog.questions = buildThreeStageQuestions(chapter.words, chapter.words);
        }
        prog.mode = 'assessment';
        saveState();
      }
      currentRoute.view = 'chapter';
      currentRoute.chapterId = chapterId;
      render();
    }

    function openErrorNotebook() {
      if ('speechSynthesis' in window) window.speechSynthesis.cancel();
      currentRoute.view = 'notebook';
      render();
    }

    function updateGlobalBadges() {
      const errorWordsCount = Object.values(appState.wordStats).filter(s => s.totalErrors > 0).length;
      const badge = document.getElementById('nav-error-count');
      if (badge) badge.textContent = errorWordsCount;

      const homeBtn = document.getElementById('btn-nav-home');
      if (homeBtn) {
        homeBtn.classList.toggle('hidden', currentRoute.view === 'home');
      }
    }

    function render() {
      updateSpeedButtonsUI();
      updateGlobalBadges();
      const main = document.getElementById('app-main');

      if (currentRoute.view === 'home') {
        main.innerHTML = renderHomeView();
      } else if (currentRoute.view === 'chapter') {
        main.innerHTML = renderChapterView(currentRoute.chapterId);
      } else if (currentRoute.view === 'notebook') {
        main.innerHTML = renderErrorNotebookView();
      } else if (currentRoute.view === 'error_practice') {
        main.innerHTML = renderErrorPracticeView();
      }
    }

    function switchChapterMode(chapterId, targetMode) {
      if ('speechSynthesis' in window) window.speechSynthesis.cancel();
      const prog = ensureChapterProgress(chapterId);
      prog.mode = targetMode;
      saveState();
      render();
    }

    function confirmResetChapter(chapterId) {
      showConfirmModal(
        '重置本章进度',
        '确定要重置当前章节的单词卡浏览位置与考核答题进度吗？（注：错词本中的累计历史错误次数将予以保留，不会被清零。）',
        () => {
          const chapter = findChapterById(chapterId);
          appState.chapterProgress[chapterId] = {
            mode: 'flashcards',
            flashcardIndex: 0,
            viewedCards: [0],
            currentStage: 1,
            currentQuestionIndex: 0,
            questions: buildThreeStageQuestions(chapter.words, chapter.words)
          };
          saveState();
          render();
          showToast('本章进度已重置');
        }
      );
    }

    function goFlashcard(newIndex) {
      if ('speechSynthesis' in window) window.speechSynthesis.cancel();
      const chapter = findChapterById(currentRoute.chapterId);
      const prog = ensureChapterProgress(chapter.id);
      if (newIndex < 0 || newIndex >= chapter.words.length) return;

      prog.flashcardIndex = newIndex;
      if (!prog.viewedCards) prog.viewedCards = [];
      if (!prog.viewedCards.includes(newIndex)) {
        prog.viewedCards.push(newIndex);
      }
      saveState();
      render();
    }

    /**
     * ----------------------------------------------------------------------------
     * 子视图 2B：三阶段考核闭环（同时复用于章节考核与错词专项训练）
     * ----------------------------------------------------------------------------
     */
    function renderAssessmentSession(chapter, sessionState, isErrorPractice = false) {
      const stage = sessionState.currentStage; // 1 | 2 | 3
      const qList = sessionState.questions[stage];
      const qIdx = Math.min(sessionState.currentQuestionIndex, qList.length - 1);
      const question = qList[qIdx];
      const wordObj = findWordById(question.wordId);
      const stat = getWordErrorStat(wordObj.id);

      // 计算总体完成度
      const totalQuestionsCount = [1, 2, 3].reduce((acc, s) => acc + sessionState.questions[s].length, 0);
      const completedQuestionsCount = [1, 2, 3].reduce(
        (acc, s) => acc + sessionState.questions[s].filter(q => q.submitted).length,
        0
      );
      const progressPct = Math.round((completedQuestionsCount / totalQuestionsCount) * 100);

      const stageNames = {

    /**
     * 获取当前处于激活状态的考核会话（章节考核 或 错词专项训练）
     */
    function getActiveSession() {
      if (currentRoute.view === 'error_practice' && currentRoute.practiceSession) {
        return { session: currentRoute.practiceSession, isPractice: true };
      }
      if (currentRoute.view === 'chapter') {
        const prog = ensureChapterProgress(currentRoute.chapterId);
        return { session: prog, isPractice: false };
      }
      return null;
    }

    /**
     * 提交选择题答案（阶段 1 认词 & 阶段 3 场景应用）
     * 严格包含单题提交锁 (question.submitted)，绝不重复计数
     */
    function submitChoiceAnswer(selectedOption) {
      const active = getActiveSession();
      if (!active) return;
      const { session, isPractice } = active;
      const qList = session.questions[session.currentStage];
      const question = qList[session.currentQuestionIndex];

      // 单题防刷锁定检查
      if (question.submitted) return;

      const isCorrect = selectedOption === question.correctAnswer;
      question.submitted = true;
      question.userAnswer = selectedOption;
      question.isCorrect = isCorrect;

      // 更新全局双维度错误统计
      recordWordAttempt(question.wordId, isCorrect);

      if (!isPractice) saveState();
      render();
    }

    /**
     * 提交听音默写答案（阶段 2）
     */
    function submitDictationAnswer() {
      const active = getActiveSession();
      if (!active) return;
      const { session, isPractice } = active;
      const qList = session.questions[2];
      const question = qList[session.currentQuestionIndex];

      // 单题防刷锁定检查
      if (question.submitted) return;

      const inputEl = document.getElementById('dictation-input');
      const rawInput = inputEl ? inputEl.value : '';

      if (!rawInput.trim()) {
        showToast('请先输入英文拼写后再提交检查');
        if (inputEl) inputEl.focus();
        return;
      }

      const normalizedUser = normalizeDictationText(rawInput);
      const normalizedTarget = normalizeDictationText(question.correctAnswer);
      const isCorrect = normalizedUser === normalizedTarget;

      question.submitted = true;
      question.userAnswer = rawInput;
      question.isCorrect = isCorrect;

      recordWordAttempt(question.wordId, isCorrect);

      if (!isPractice) saveState();
      render();
    }

    function handleDictationKeydown(e) {
      if (e.key === 'Enter') {
        e.preventDefault();
        const active = getActiveSession();
        if (!active) return;
        const q = active.session.questions[2][active.session.currentQuestionIndex];
        if (!q.submitted) {
          submitDictationAnswer();
        } else {
          advanceAssessmentQuestion();
        }
      }
    }

    /**
     * 推进到下一题 / 下一阶段 / 完成结算
     */
    function advanceAssessmentQuestion() {
      if ('speechSynthesis' in window) window.speechSynthesis.cancel();
      const active = getActiveSession();
      if (!active) return;
      const { session, isPractice } = active;

      const stage = session.currentStage;
      const qList = session.questions[stage];

      if (session.currentQuestionIndex < qList.length - 1) {
        session.currentQuestionIndex += 1;
      } else if (stage < 3) {
        session.currentStage += 1;
        session.currentQuestionIndex = 0;
      } else {
        // 全部三阶段已完成
        if (isPractice) {
          session.completed = true;
        } else {
          session.mode = 'summary';
          completeChapterAndUnlockNext(currentRoute.chapterId);
        }
      }

      if (!isPractice) saveState();
      render();

      // 若进入阶段 2 默写且尚未作答，自动聚焦输入框
      setTimeout(() => {
        const inputEl = document.getElementById('dictation-input');
        if (inputEl && !inputEl.disabled) inputEl.focus();
      }, 50);
    }

    /**
     * 标记当前章节通关并解锁下一章（不限正确率）
     */
    function completeChapterAndUnlockNext(chapterId) {
      if (!appState.completedChapters.includes(chapterId)) {
        appState.completedChapters.push(chapterId);
      }
      const idx = CHAPTERS_DATA.findIndex(c => c.id === chapterId);
      if (idx !== -1 && idx + 1 < CHAPTERS_DATA.length) {
        const nextChId = CHAPTERS_DATA[idx + 1].id;
        if (!appState.unlockedChapters.includes(nextChId)) {
          appState.unlockedChapters.push(nextChId);
        }
      }
      saveState();
    }

    /**

     * 子视图 2C：章节通关结算页 (CHAPTER SUMMARY)
     * ----------------------------------------------------------------------------
     */

          else wrongWordIds.add(q.wordId);

      const accuracy = totalQ > 0 ? Math.round((correctQ / totalQ) * 100) : 100;
      const idx = CHAPTERS_DATA.findIndex(c => c.id === chapter.id);
      const nextChapter = idx + 1 < CHAPTERS_DATA.length ? CHAPTERS_DATA[idx + 1] : null;

      return `
        <div class="fade-in academic-card rounded-2xl p-6 sm:p-8 text-center space-y-6">
          <div class="w-16 h-16 rounded-2xl bg-emerald-100 text-emerald-700 flex items-center justify-center text-3xl mx-auto">
            🎓
          </div>
          <div>
            <span class="inline-block px-3 py-1 rounded-full bg-emerald-50 text-emerald-700 border border-emerald-200 text-xs font-bold mb-2">
              ${chapter.number} · 考核流程已完成
            </span>
            <h2 class="text-2xl sm:text-3xl font-bold text-slate-900">

              <p class="text-sm text-brand-800 font-medium mt-2">
                🔓 已自动为你解锁下一章：<strong>${nextChapter.number} · ${nextChapter.title}</strong>
              </p>
            ` : `
              <p class="text-sm text-slate-600 mt-2">你已完成当前词库中的所有章节学习！</p>
            `}
          </div>

          <div class="grid grid-cols-3 gap-3 max-w-md mx-auto">
            <div class="p-3.5 rounded-xl bg-slate-50 border border-slate-200">
              <div class="text-xl font-bold text-slate-900">${accuracy}%</div>
              <div class="text-xs text-slate-500 mt-0.5">本轮正确率</div>
            </div>
            <div class="p-3.5 rounded-xl bg-emerald-50/70 border border-emerald-200">
              <div class="text-xl font-bold text-emerald-700">${correctQ}/${totalQ}</div>
              <div class="text-xs text-emerald-700 mt-0.5">答对题数</div>

            </div>
          </div>

          <div class="flex flex-wrap items-center justify-center gap-3 pt-2">

              <button onclick="openChapterErrorFilter('${chapter.id}')"
                class="px-5 py-3 rounded-xl bg-rose-50 hover:bg-rose-100 text-rose-700 border border-rose-200 text-sm font-bold transition">
                前往错词本复习本章错词 (${wrongWordIds.size})
              </button>
            ` : ''}

            ${nextChapter ? `
              <button onclick="openChapter('${nextChapter.id}', 'flashcards')"
                class="px-6 py-3 rounded-xl bg-brand-900 hover:bg-brand-800 text-white text-sm font-bold shadow-sm transition">
                进入 ${nextChapter.number} 单词卡 →
              </button>
            ` : `
              <button onclick="navigateHome()"
                class="px-6 py-3 rounded-xl bg-brand-900 hover:bg-brand-800 text-white text-sm font-bold shadow-sm transition">
                返回章节目录
              </button>
            `}
          </div>
        </div>
      `;
    }

    function getFilteredErrorWords() {
      const allWords = CHAPTERS_DATA.flatMap(c => c.words);
      return allWords
        .filter(w => {
          const stat = getWordErrorStat(w.id);
          if (stat.totalErrors <= 0) return false;
          if (currentRoute.notebookFilter !== 'all' && w.chapterId !== currentRoute.notebookFilter) {
            return false;
          }
          return true;
        })
        // 按累计错误次数 totalErrors 从高到低排序
        .sort((a, b) => getWordErrorStat(b.id).totalErrors - getWordErrorStat(a.id).totalErrors);
    }

          }).join('');

                按累计错误次数由高到低排列（答对后不会清零历史累计错误次数，便于长期追踪高频卡点）
              </p>

    /**
     * ----------------------------------------------------------------------------
     * 视图 4：错词专项三阶段训练 (ERROR NOTEBOOK 3-STAGE PRACTICE)
     * ----------------------------------------------------------------------------
     */
    function startErrorPractice() {
      const targetWords = getFilteredErrorWords();
      if (targetWords.length === 0) return;

      // 干扰项池：若筛选了某一章则优先用该章词库，否则用全库词池
      const poolWords = currentRoute.notebookFilter === 'all'
        ? CHAPTERS_DATA.flatMap(c => c.words)
        : findChapterById(currentRoute.notebookFilter).words;

      currentRoute.practiceSession = {
        currentStage: 1,
        currentQuestionIndex: 0,
        completed: false,
        questions: buildThreeStageQuestions(targetWords, poolWords)
      };
      currentRoute.view = 'error_practice';
      render();
    }

    function renderErrorPracticeView() {
      const session = currentRoute.practiceSession;
      if (!session) {
        openErrorNotebook();
        return '';
      }

      const topBar = `
        <div class="flex items-center justify-between gap-3 mb-5">
          <button onclick="openErrorNotebook()" class="text-sm text-slate-500 hover:text-brand-900 font-medium transition">
            ← 退出训练并返回错词本
          </button>
          <span class="text-xs font-bold text-rose-700 bg-rose-50 px-3 py-1 rounded-full border border-rose-200">
            错词专项闭环重练（认词 → 默写 → 场景应用）
          </span>
        </div>
      `;

      if (session.completed) {
        let totalQ = 0;
        let correctQ = 0;
        [1, 2, 3].forEach(s => {
          session.questions[s].forEach(q => {
            totalQ += 1;
            if (q.isCorrect) correctQ += 1;
          });
        });

        return topBar + `
          <div class="fade-in academic-card rounded-2xl p-8 text-center space-y-5">
            <div class="text-4xl">🏆</div>
            <h2 class="text-2xl font-bold text-slate-900">错词专项三阶段训练完成！</h2>
            <p class="text-sm text-slate-600">
              本轮共完成 <strong>${totalQ}</strong> 道强化练习，答对 <strong class="text-emerald-700">${correctQ}</strong> 题。答对词汇的“当前连错次数”已清零。
            </p>
            <div class="pt-2">
              <button onclick="openErrorNotebook()" class="px-6 py-3 rounded-xl bg-brand-900 hover:bg-brand-800 text-white text-sm font-bold transition">
                返回错词本查看最新统计
              </button>
            </div>
          </div>
        `;
      }

      return topBar + renderAssessmentSession(null, session, true);
    }

    /**
     * ============================================================================
     * 七、 工具函数与键盘快捷键支持
     * ============================================================================
     */
    function escapeHtml(str) {
      if (str === null || str === undefined) return '';
      return String(str)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;')
        .replace(/'/g, '&#39;');
    }

    function escapeJsStr(str) {
      if (!str) return '';
      return String(str).replace(/\\/g, '\\\\').replace(/'/g, "\\'");
    }

    let toastTimer = null;
    function showToast(msg) {
      const box = document.getElementById('toast-box');
      const text = document.getElementById('toast-msg');
      if (!box || !text) return;
      text.textContent = msg;
      box.classList.remove('opacity-0');
      clearTimeout(toastTimer);
      toastTimer = setTimeout(() => {
        box.classList.add('opacity-0');
      }, 2400);
    }

    function showConfirmModal(title, desc, onConfirm) {
      const overlay = document.getElementById('modal-overlay');
      document.getElementById('modal-title').textContent = title;
      document.getElementById('modal-desc').textContent = desc;
      overlay.classList.remove('hidden');
      overlay.classList.add('flex');

      const close = () => {
        overlay.classList.add('hidden');
        overlay.classList.remove('flex');
      };

      document.getElementById('modal-cancel').onclick = close;
      document.getElementById('modal-confirm').onclick = () => {
        close();
        onConfirm();
      };
    }

    // 全局键盘辅助：单词卡左右键切换、考核已提交时按 Enter 下一题
    document.addEventListener('keydown', (e) => {
      if (currentRoute.view === 'chapter') {
        const prog = ensureChapterProgress(currentRoute.chapterId);
        if (prog.mode === 'flashcards') {
          if (e.key === 'ArrowLeft') goFlashcard(prog.flashcardIndex - 1);
          if (e.key === 'ArrowRight') goFlashcard(prog.flashcardIndex + 1);
        } else if (prog.mode === 'assessment' && e.key === 'Enter' && document.activeElement?.id !== 'dictation-input') {
          const q = prog.questions[prog.currentStage][prog.currentQuestionIndex];
          if (q && q.submitted) {
            e.preventDefault();
            advanceAssessmentQuestion();
          }
        }
      } else if (currentRoute.view === 'error_practice' && e.key === 'Enter' && document.activeElement?.id !== 'dictation-input') {
        const session = currentRoute.practiceSession;
        if (session && !session.completed) {
          const q = session.questions[session.currentStage][session.currentQuestionIndex];
          if (q && q.submitted) {
            e.preventDefault();
            advanceAssessmentQuestion();
          }
        }
      }
    });

    // 初始化首次渲染
    window.addEventListener('DOMContentLoaded', () => {
      render();
    });
  </script>
